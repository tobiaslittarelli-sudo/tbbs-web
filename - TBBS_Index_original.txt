    function crearImagen(url, clase, nombre, inicial) {
      const tipo = clase === "imagen-jugador" ? "jugador" : "equipo";

      const reemplazo = document.createElement("div");
      reemplazo.className = "sin-imagen " + tipo;
      reemplazo.textContent = inicial;

      if (!url || typeof url !== "string") {
        return reemplazo;
      }

      const imagen = document.createElement("img");
      imagen.className = clase;
      imagen.alt = nombre || "Imagen";
      imagen.loading = "eager";
      imagen.decoding = "async";

      imagen.onerror = function() {
        console.error("No se pudo cargar la imagen:", nombre, url);
        imagen.replaceWith(reemplazo);
      };

      imagen.onload = function() {
        console.log("Imagen cargada correctamente:", nombre);
      };

      imagen.src = url.trim();
      return imagen;
    }

    function crearTarjetaEquipo(equipo, indice) {
      const tarjeta = document.createElement("article");
      tarjeta.className = "tarjeta";

      const inicial = equipo.nombre
        ? equipo.nombre.trim().charAt(0).toUpperCase()
        : "E";

      tarjeta.appendChild(
        crearImagen(equipo.logo, "imagen-equipo", equipo.nombre, inicial)
      );

      const contenido = document.createElement("div");
      contenido.className = "contenido-tarjeta";

      const etiqueta = document.createElement("span");
      etiqueta.className = "etiqueta";
      etiqueta.textContent = "EQUIPO " + String(indice + 1).padStart(2, "0");

      const titulo = document.createElement("h3");
      titulo.textContent = equipo.nombre || "Equipo sin nombre";

      const descripcion = document.createElement("p");
      descripcion.textContent = "Temporada 2027";

      contenido.appendChild(etiqueta);
      contenido.appendChild(titulo);
      contenido.appendChild(descripcion);
      tarjeta.appendChild(contenido);

      return tarjeta;
    }

    function crearTarjetaJugador(jugador) {
      const tarjeta = document.createElement("article");
      tarjeta.className = "tarjeta";

      const inicial = jugador.nombre
        ? jugador.nombre.trim().charAt(0).toUpperCase()
        : "J";

      tarjeta.appendChild(
        crearImagen(jugador.foto, "imagen-jugador", jugador.nombre, inicial)
      );

      const contenido = document.createElement("div");
      contenido.className = "contenido-tarjeta";

      const etiqueta = document.createElement("span");
      etiqueta.className = "etiqueta";
      etiqueta.textContent = "JUGADOR";

      const titulo = document.createElement("h3");
      titulo.textContent = jugador.nombre || "Jugador sin nombre";

      const equipo = document.createElement("p");
      equipo.textContent = "Equipo: " + (jugador.equipo || "-");

      const puntos = document.createElement("p");
      puntos.textContent = "Puntos: " + (jugador.puntos || 0);

      const partidos = document.createElement("p");
      partidos.textContent = "Partidos: " + (jugador.partidos || 0);

      const promedio = document.createElement("p");
      promedio.textContent =
        "Promedio: " + (Number(jugador.promedio) || 0).toFixed(1);

      contenido.appendChild(etiqueta);
      contenido.appendChild(titulo);
      contenido.appendChild(equipo);
      contenido.appendChild(puntos);
      contenido.appendChild(partidos);
      contenido.appendChild(promedio);

      tarjeta.appendChild(contenido);
      return tarjeta;
    }

    function cargarDatos(datos) {
      if (
        !datos ||
        !Array.isArray(datos.equipos) ||
        !Array.isArray(datos.jugadores)
      ) {
        mostrarError({
          message: "Los datos recibidos no tienen el formato esperado."
        });
        return;
      }

      const listaEquipos = document.getElementById("listaEquipos");
      const listaJugadores = document.getElementById("listaJugadores");
      const tabla = document.getElementById("tablaEstadisticas");

      listaEquipos.innerHTML = "";
      listaJugadores.innerHTML = "";
      tabla.innerHTML = "";

      if (datos.equipos.length === 0) {
        listaEquipos.innerHTML =
          '<p class="estado">No hay equipos cargados.</p>';
      } else {
        datos.equipos.forEach(function(equipo, indice) {
          listaEquipos.appendChild(crearTarjetaEquipo(equipo, indice));
        });
      }

      if (datos.jugadores.length === 0) {
        listaJugadores.innerHTML =
          '<p class="estado">No hay jugadores cargados.</p>';

        tabla.innerHTML =
          '<tr><td colspan="5" class="estado">Todavía no hay estadísticas.</td></tr>';
        return;
      }

      datos.jugadores.forEach(function(jugador) {
        listaJugadores.appendChild(crearTarjetaJugador(jugador));

        const fila = document.createElement("tr");
        const valores = [
          jugador.nombre || "-",
          jugador.equipo || "-",
          Number(jugador.puntos) || 0,
          Number(jugador.partidos) || 0,
          (Number(jugador.promedio) || 0).toFixed(1)
        ];

        valores.forEach(function(valor) {
          const celda = document.createElement("td");
          celda.textContent = valor;
          fila.appendChild(celda);
        });

        tabla.appendChild(fila);
      });

      console.log("Datos cargados:", datos);
    }

    function mostrarError(error) {
      const mensaje = error && error.message
        ? error.message
        : String(error);

      document.getElementById("listaEquipos").innerHTML =
        '<p class="estado error">Error: ' + mensaje + '</p>';

      document.getElementById("listaJugadores").innerHTML =
        '<p class="estado error">Error: ' + mensaje + '</p>';

      document.getElementById("tablaEstadisticas").innerHTML =
        '<tr><td colspan="5" class="estado error">' + mensaje + '</td></tr>';

      console.error("Error T.B.B.S.:", error);
    }

    function iniciarCarga() {
      if (
        typeof google === "undefined" ||
        !google.script ||
        !google.script.run
      ) {
        mostrarError({
          message: "No se pudo conectar con Google Apps Script."
        });
        return;
      }

      google.script.run
        .withSuccessHandler(cargarDatos)
        .withFailureHandler(mostrarError)
        .obtenerDatos();
    }

    window.addEventListener("load", iniciarCarga);
